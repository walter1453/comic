        let currentChapterIndex = 0;
        const chapterSelect = document.getElementById("chapter-select");
        const readerContainer = document.getElementById("reader-container");
        const comicTitle = document.getElementById("comic-title");
        const comicDesc = document.getElementById("comic-desc");
        const prevBtn = document.getElementById("prev-btn");
        const nextBtn = document.getElementById("next-btn");

        function init() {
            chapters.forEach((chapter, index) => {
                const option = document.createElement("option");
                option.value = index;
                option.textContent = chapter.title;
                chapterSelect.appendChild(option);
            });
            renderChapter(0);
        }

        function renderChapter(index) {
            currentChapterIndex = index;
            chapterSelect.value = index;
            
            const chapter = chapters[index];
            comicTitle.textContent = chapter.title;
            comicDesc.textContent = chapter.description;
            
            readerContainer.innerHTML = "";
            
            chapter.pages.forEach(pageSrc => {
                const img = document.createElement("img");
                img.src = pageSrc;
                img.alt = chapter.title;
                img.loading = "lazy";
                readerContainer.appendChild(img);
            });

            prevBtn.disabled = index === 0;
            nextBtn.disabled = index === chapters.length - 1;
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function loadChapter() {
            renderChapter(parseInt(chapterSelect.value));
        }

        function prevChapter() {
            if (currentChapterIndex > 0) renderChapter(currentChapterIndex - 1);
        }

        function nextChapter() {
            if (currentChapterIndex < chapters.length - 1) renderChapter(currentChapterIndex + 1);
        }

        function toggleTheme() {
            document.body.classList.toggle("light-theme");
            const btn = document.getElementById("theme-toggle");
            btn.textContent = document.body.classList.contains("light-theme") ? "Темная тема" : "Светлая тема";
        }

        init();
